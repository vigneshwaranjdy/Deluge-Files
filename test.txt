// Explanation:
// An Armstrong number (also known as a narcissistic number) is a number
// that is equal to the sum of its own digits, each raised to the power of the
// number of digits.
// Example: 153 : (1^3 + 5^3 + 3^3 = 1 + 125 + 27 =153)

size = n.size();
i = 0;

for each no in n
{
	nu = no.toNumber();
	var = nu.power(size);
	info nu +"^"+size+" = " +  var ;
	i = i + var;
}
// info i;

if (i == num) 
{
	info "total = " + i ;
	info "The given no is Armstrong" ;
}

else 
{
	info "total = " + i ;
	info "The given no is not an Armstrong" ;
}
